Get Started in 4 Lines

Assuming your GPU drivers work. No guarantees.

run_qwen.py
from transformers import AutoModelForCausalLM, AutoTokenizer

# Yes, it's really this easy. No, we're not hiding anything.
model_id = "Qwen/Qwen3.7-30B-Open"

tokenizer = AutoTokenizer.from_pretrained(model_id)
model = AutoModelForCausalLM.from_pretrained(
    model_id,
    torch_dtype="auto",
    device_map="auto",  # or "cpu" if you enjoy suffering
)

messages = [
    {"role": "user", "content": "Explain quantum computing like I'm a golden retriever"}
]

inputs = tokenizer.apply_chat_template(
    messages, tokenize=True, add_generation_prompt=True, return_tensors="pt"
).to(model.device)

outputs = model.generate(inputs, max_new_tokens=512, temperature=0.7)
print(tokenizer.decode(outputs[0], skip_special_tokens=True))
# Output: "Okay hooman, imagine qubits as tennis balls..."
