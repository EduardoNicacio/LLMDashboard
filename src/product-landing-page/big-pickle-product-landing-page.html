Usage

One line to run them all

Compatible with Hugging Face Transformers. No registry keys. No prayer circles required.

transformers_example.py
# pip install transformers>=4.50.0 accelerate from transformers import AutoModelForCausalLM, AutoTokenizer model = AutoModelForCausalLM.from_pretrained( "Qwen/Qwen3.7-30B-Open", torch_dtype="bfloat16", device_map="auto", # trust_remote_code=True # we pinky promise it's safe ) tokenizer = AutoTokenizer.from_pretrained("Qwen/Qwen3.7-30B-Open") prompt = "Write a haiku about open-source AI" inputs = tokenizer(prompt, return_tensors="pt").to("cuda") output = model.generate( **inputs, max_new_tokens=512, temperature=0.7, # do_sample=True, # live dangerously ) print(tokenizer.decode(output[0], skip_special_tokens=True)) # > Open weights shimmer # > GPUs humming in the dark # > Finally, it's here
