Quick Start
Up and running in 3 lines
Copy. Paste. Pretend you wrote it yourself.
python
from transformers import AutoModelForCausalLM, AutoTokenizer
 
model_id = "Qwen/Qwen3.7-30B-Open"
 
tokenizer = AutoTokenizer.from_pretrained(model_id)
model = AutoModelForCausalLM.from_pretrained(
    model_id,
    torch_dtype="bfloat16",
    device_map="auto", # pray to your VRAM gods
)
 
messages = [{"role": "user", "content": "Explain quantum computing to a 5 year old"}]
 
inputs = tokenizer.apply_chat_template(messages, return_tensors="pt")
outputs = model.generate(inputs, max_new_tokens=512)
 
print(tokenizer.decode(outputs[0], skip_special_tokens=True))
