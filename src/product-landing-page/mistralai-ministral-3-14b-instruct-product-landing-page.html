Get Started with Transformers

Here's how to load Qwen3.7-Open in your Python environment:

# First install the required packages (if you have enough VRAM)
pip install torch transformers accelerate bitsandbytes

# Then load the model
from transformers import AutoModelForCausalLM, AutoTokenizer

model = AutoModelForCausalLM.from_pretrained(
    "Qwen/Qwen3.7-30B-Open",
    device_map="auto",  # This will use all your GPUs (and possibly your CPU too)
    torch_dtype=torch.bfloat16,  # Or float16 if you're feeling brave
    trust_remote_code=True,
    offload_folder="disk"  # For when you run out of memory
)

tokenizer = AutoTokenizer.from_pretrained("Qwen/Qwen3.7-30B-Open")

# Example usage (after waiting several hours for loading)
inputs = tokenizer("Explain quantum computing to a 5-year-old", return_tensors="pt").to(model.device)
outputs = model.generate(**inputs, max_new_tokens=1024)

print(tokenizer.decode(outputs[0], skip_special_tokens=True))

# Pro tip: Add --deepspeed to your command line for maximum drama
