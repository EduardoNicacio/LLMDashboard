Quickstart

Three lines, one coffee, no excuses.

Works with stock transformers ≥ 4.43. If you get an error, the error is part of the experience.

qwen3.7-30b-open.py 
# pip install transformers>=4.43 torch accelerate
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

# yes, it really is named this
model_id = "qwen/Qwen3.7-30B-Open"

tok = AutoTokenizer.from_pretrained(model_id)
model = AutoModelForCausalLM.from_pretrained(
    model_id,
    torch_dtype=torch.bfloat16,
    device_map="auto",          # or "dual-3090"
    attn_implementation="flash_attention_2",
)

prompt = "Write a haiku about finally getting the open weights."
inputs = tok(prompt, return_tensors="pt").to(model.device)

out = model.generate(
    **inputs,
    max_new_tokens=128,
    do_sample=True,
    temperature=0.7,
    top_p=0.9,
)
print(tok.decode(out[0], skip_special_tokens=True))
# >> weights arrive / the model dreams in public / refreshing stops
